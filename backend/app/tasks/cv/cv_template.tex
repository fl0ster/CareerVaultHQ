% -----------------------------------------------------------------------------
% Macros

% \cvsection{Title}: upper-case heading glued to whatever follows it.
\newcommand{\cvsectionheading}[1]{%
  {\color{accent}\fontsize{12.5pt}{15pt}\selectfont\bfseries\MakeUppercase{#1}\par}%
  \nobreak\vspace{2pt}%
  \nobreak\vspace{\cvSectionAfter}\nobreak
  \global\cvfirstroletrue
}
\newcommand{\cvsection}[1]{%
  \par\vspace{\cvSectionBefore}%
  \cvsectionheading{#1}%
}

% cvsectionblock environment: heading AND content in one minipage, so a short section
% (Education, Technologies, Languages) is never split across pages.
\newenvironment{cvsectionblock}[1]{%
  \par\vspace{\cvSectionBefore}%
  \begin{minipage}{\linewidth}\raggedright
  \cvsectionheading{#1}%
}{%
  \end{minipage}\par
}

% \cvitem{Label}{Text}: one labelled line, e.g. for Languages / Technologies.
\newcommand{\cvitem}[2]{{\bfseries #1:} #2\par\vspace{2pt}}

% \cvtech{...}: "Tech:" line used inside a role.
\newcommand{\cvtech}[1]{%
  \par\vspace{\cvTechGap}%
  {\color{muted}\small\textbf{Tech:} #1\par}%
}

% cvrole environment
%   \begin{cvrole}{Role title}{Company}{Dates}{Location}
%     \item ...
%     \cvtech{...}          % optional
%   \end{cvrole}
% The block is typeset into a box first; if it does not fit on the remaining page a new
% page is started and the hairline separator above it is omitted.
\newcommand{\cvroleheader}[4]{%
  {\fontsize{12pt}{14pt}\selectfont\bfseries #1}%
  \hfill{\color{muted}\small #3}\par
  \vspace{1pt}%
  {\color{muted}\itshape #2\ifx\relax#4\relax\else\ \textperiodcentered\ #4\fi\par}%
  \vspace{\cvRoleHeadGap}%
}
\newif\ifcvfirstrole
\newsavebox{\cvrolebox}
\newlength{\cvroleneed}
\newlength{\cvroleavail}
\newenvironment{cvrole}[4]{%
  \begin{lrbox}{\cvrolebox}%
  \begin{minipage}{\linewidth}%
    \raggedright
    \cvroleheader{#1}{#2}{#3}{#4}%
    \begin{itemize}
}{%
    \end{itemize}%
  \end{minipage}%
  \end{lrbox}%
  \par\penalty0
  \ifcvfirstrole
    \global\cvfirstrolefalse
    \vspace{\cvRoleGap}\usebox{\cvrolebox}\par
  \else
    \setlength{\cvroleneed}{\ht\cvrolebox}%
    \addtolength{\cvroleneed}{\dp\cvrolebox}%
    \addtolength{\cvroleneed}{2\cvRoleGap}%
    \addtolength{\cvroleneed}{\baselineskip}% rule line + slack
    \setlength{\cvroleavail}{\pagegoal}%
    \addtolength{\cvroleavail}{-\pagetotal}%
    \ifdim\cvroleneed>\cvroleavail
      \newpage\usebox{\cvrolebox}\par
    \else
      \vspace{\cvRoleGap}\cvrule\usebox{\cvrolebox}\par
    \fi
  \fi
}

% \cvblock: generic unbreakable block (consent, ...).
\newenvironment{cvblock}{\par\begin{minipage}{\linewidth}\raggedright}{\end{minipage}\par}

% Hairline drawn above every role except the first one in a section.
\newcommand{\cvrule}{{\color{rule}\rule{\linewidth}{0.4pt}}\par\vspace{\cvRoleGap}}
